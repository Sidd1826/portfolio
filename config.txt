"""
config.py — Central config.

On Cloud Run: secrets are pulled from GCP Secret Manager at startup.
Locally: values are loaded from .env via python-dotenv.

Secret Manager is only used when GOOGLE_CLOUD_PROJECT is set (i.e. on GCP).
Locally it falls back to .env / os.environ as before — no GCP credentials needed.
"""
from __future__ import annotations

import os

# Load .env for local dev (no-op on Cloud Run)
try:
    from dotenv import load_dotenv
    load_dotenv(override=False)
except ImportError:
    pass


def _env(name: str, default: str = "") -> str:
    return os.getenv(name, default).strip()


def _env_float(name: str, default: float) -> float:
    raw = os.getenv(name)
    return float(raw) if raw else default


def _env_int(name: str, default: int) -> int:
    raw = os.getenv(name)
    return int(raw) if raw else default


# ---------------------------------------------------------------------------
# Secret Manager — only runs on GCP (when GOOGLE_CLOUD_PROJECT is set).
# Pulls each secret by name and injects into os.environ so the rest of
# config.py can read them with plain os.getenv() calls — no other file changes.
# ---------------------------------------------------------------------------

def _load_secrets_from_gcp() -> None:
    """Pull all secrets from Secret Manager and set them into os.environ."""
    project_id = os.getenv("GOOGLE_CLOUD_PROJECT")
    if not project_id:
        # Not on GCP — use .env / os.environ as-is
        return

    try:
        from google.cloud import secretmanager
    except ImportError:
        raise RuntimeError(
            "google-cloud-secret-manager is not installed. "
            "Run: pip install google-cloud-secret-manager"
        )

    client = secretmanager.SecretManagerServiceClient()

    # Map: env var name -> Secret Manager secret name
    # The secret name must match exactly what you created in Secret Manager.
    secrets_to_load = {
        "UPI_USERNAME":                           "UPI_USERNAME",
        "UPI_PASSWORD":                           "UPI_PASSWORD",
        "UPI_AES_KEY_HEX":                        "UPI_AES_KEY_HEX",
        "HSBC_TRUST_TOKEN_TRANSACTION_HISTORY":   "HSBC_TRUST_TOKEN_TRANSACTION_HISTORY",
        "HSBC_TRUST_TOKEN_APPLY_HOLD":            "HSBC_TRUST_TOKEN_APPLY_HOLD",
        "HSBC_TRUST_TOKEN_DEMAND_DEPOSIT":        "HSBC_TRUST_TOKEN_DEMAND_DEPOSIT",
    }

    for env_var, secret_name in secrets_to_load.items():
        # Skip if already set (allows override via --set-env-vars if ever needed)
        if os.environ.get(env_var):
            continue
        try:
            secret_path = f"projects/{project_id}/secrets/{secret_name}/versions/latest"
            response = client.access_secret_version(request={"name": secret_path})
            os.environ[env_var] = response.payload.data.decode("utf-8").strip()
        except Exception as e:
            # Raise loudly — a missing secret should crash startup, not silently
            # run with an empty value which causes confusing downstream errors.
            raise RuntimeError(
                f"Failed to load secret '{secret_name}' from Secret Manager: {e}"
            ) from e


# Pull secrets before reading any config values below
_load_secrets_from_gcp()
