"""
config.py — Central config.

On Cloud Run: secrets are pulled from GCP Secret Manager at startup,
              BigQuery uses Application Default Credentials (ADC).
Locally:      values are loaded from .env via python-dotenv.

Secret Manager is only contacted when GOOGLE_CLOUD_PROJECT is set (i.e. on
GCP). Locally it falls back to .env / os.environ — no GCP credentials needed.
"""
from __future__ import annotations

import os

try:
    from dotenv import load_dotenv
    load_dotenv(override=False)
except ImportError:
    pass


def _env(name: str, default: str = "") -> str:
    return os.getenv(name, default).strip()


def _env_float(name: str, default: float) -> float:
    raw = os.getenv(name)
    return float(raw) if raw else default


def _env_int(name: str, default: int) -> int:
    raw = os.getenv(name)
    return int(raw) if raw else default


# ---------------------------------------------------------------------------
# Secret Manager — only runs on GCP (when GOOGLE_CLOUD_PROJECT is set).
# ---------------------------------------------------------------------------

def _load_secrets_from_gcp() -> None:
    project_id = os.getenv("GOOGLE_CLOUD_PROJECT")
    if not project_id:
        return

    try:
        from google.cloud import secretmanager
    except ImportError:
        raise RuntimeError(
            "google-cloud-secret-manager is not installed. "
            "Run: pip install google-cloud-secret-manager"
        )

    client = secretmanager.SecretManagerServiceClient()

    secrets_to_load = {
        "UPI_USERNAME":              "UPI_USERNAME",
        "UPI_PASSWORD":              "UPI_PASSWORD",
        "UPI_AES_KEY_HEX":           "UPI_AES_KEY_HEX",
        "HSBC_TOKEN_USERNAME":       "HSBC_TOKEN_USERNAME",
        "HSBC_TOKEN_PASSWORD":       "HSBC_TOKEN_PASSWORD",
    }

    for env_var, secret_name in secrets_to_load.items():
        if os.environ.get(env_var):
            continue
        try:
            secret_path = f"projects/{project_id}/secrets/{secret_name}/versions/latest"
            response = client.access_secret_version(request={"name": secret_path})
            os.environ[env_var] = response.payload.data.decode("utf-8").strip()
        except Exception as e:
            raise RuntimeError(
                f"Failed to load secret '{secret_name}' from Secret Manager: {e}"
            ) from e


_load_secrets_from_gcp()
