"""
config.py — Central config. Everything comes from env vars so Cloud Run
revisions can set them via Secret Manager / --set-env-vars.

Locally, values are loaded from a .env file via python-dotenv.
On Cloud Run, os.environ is already populated — dotenv is a no-op.
"""
from __future__ import annotations

import os

try:
    from dotenv import load_dotenv
    load_dotenv(override=False)
except ImportError:
    pass


def _env(name: str, default: str = "") -> str:
    return os.getenv(name, default).strip()


def _env_float(name: str, default: float) -> float:
    raw = os.getenv(name)
    return float(raw) if raw else default


def _env_int(name: str, default: int) -> int:
    raw = os.getenv(name)
    return int(raw) if raw else default
