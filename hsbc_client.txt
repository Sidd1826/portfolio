"""
hsbc_client.py - HSBC Hub API client using `requests` (sync).

Why `requests` instead of `httpx`?
  Corporate proxies + self-signed certs work reliably with `requests` and
  `verify=False`. httpx's proxy+SSL combination is more fragile in some
  versions and was causing CERTIFICATE_VERIFY_FAILED errors.

FastAPI handles these sync calls by running them in a threadpool —
no extra setup needed; just `def` (not `async def`) endpoints would
also work, but we wrap each call in `asyncio.to_thread(...)` from the
routes to keep the event loop free.

Every request:
  - uses the configured proxy (config.PROXY_URL)
  - sets verify=False
  - has a 60s timeout

Headers are defined directly inside each function — no env vars,
no caller headers, no confusion.
"""
from __future__ import annotations

import uuid
import warnings
from typing import Any, Optional

import requests
from requests.adapters import HTTPAdapter
from urllib3.exceptions import InsecureRequestWarning
from urllib3.util.retry import Retry

import config
from logger import get_logger

# Silence "InsecureRequestWarning" since we intentionally use verify=False
warnings.filterwarnings("ignore", category=InsecureRequestWarning)

log = get_logger(__name__)


class HSBCError(Exception):
    def __init__(self, msg: str, status_code: int = 502, upstream_status: Optional[int] = None):
        super().__init__(msg)
        self.status_code     = status_code
        self.upstream_status = upstream_status


class HSBCHubClient:
    """Sync HSBC Hub client. Built once and reused across requests (connection pooling)."""

    def __init__(self) -> None:
        # Pooled session with auto-retry on transient failures
        self._session = requests.Session()

        retry = Retry(
            total=2,
            backoff_factor=0.5,
            status_forcelist=(429, 500, 502, 503, 504),
            allowed_methods=frozenset(["GET", "POST"]),
            raise_on_status=False,
        )
        adapter = HTTPAdapter(pool_connections=20, pool_maxsize=50, max_retries=retry)
        self._session.mount("https://", adapter)
        self._session.mount("http://",  adapter)

        # Proxy applied to every call from this session
        if config.PROXY_URL:
            self._session.proxies = {"http": config.PROXY_URL, "https": config.PROXY_URL}
            log.info("HSBCHubClient: PROXY set to %s", config.PROXY_URL)
        else:
            log.info("HSBCHubClient: NO proxy configured")

        # verify=False applied via session.verify
        self._session.verify = False
        self._timeout = (config.HTTP_CONNECT_TIMEOUT, config.HTTP_READ_TIMEOUT)
        log.info("HSBCHubClient: verify=False, timeout=%s", self._timeout)

    def close(self) -> None:
        self._session.close()

    # ------------------------------------------------------------------ #
    # Internal helpers
    # ------------------------------------------------------------------ #
    def _get(self, url: str, headers: dict[str, str], params: dict[str, str], label: str) -> dict[str, Any]:
        try:
            resp = self._session.get(url, headers=headers, params=params, timeout=self._timeout)
        except requests.RequestException as e:
            log.exception("%s network error", label)
            raise HSBCError(f"{label} network error: {type(e).__name__}: {e}") from e
        log.info("%s response: HTTP %d", label, resp.status_code)
        return self._parse(resp, label)

    def _post(self, url: str, headers: dict[str, str], body: dict[str, Any], label: str) -> dict[str, Any]:
        try:
            resp = self._session.post(url, headers=headers, json=body, timeout=self._timeout)
        except requests.RequestException as e:
            log.exception("%s network error", label)
            raise HSBCError(f"{label} network error: {type(e).__name__}: {e}") from e
        log.info("%s response: HTTP %d", label, resp.status_code)
        return self._parse(resp, label)

    @staticmethod
    def _parse(resp: requests.Response, label: str) -> dict[str, Any]:
        if resp.status_code not in (200, 201):
            log.error("%s failed HTTP %d | body=%s", label, resp.status_code, resp.text[:400])
            raise HSBCError(f"{label} returned HTTP {resp.status_code}", upstream_status=resp.status_code)
        try:
            return resp.json()
        except ValueError:
            return {"raw": resp.text}
