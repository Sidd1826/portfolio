"""
hsbc_client.py — Sync requests client for HSBC Hub APIs.

Three endpoints, each with its own hardcoded header set (per the curls)
and its own E2E trust token (loaded from config).

verify=False is intentional — HSBC internal hosts use an internal CA.
No proxy — these are internal HSBC endpoints reachable directly.
"""

# ---------------------------------------------------------------------------
# Errors
# ---------------------------------------------------------------------------

class HSBCError(Exception):
    def __init__(self, message: str, *, status_code: int = 502, upstream_status: Optional[int] = None, upstream_body: Any = None):
        super().__init__(message)
        self.status_code = status_code
        self.upstream_status = upstream_status
        self.upstream_body = upstream_body


# ---------------------------------------------------------------------------
# Token manager — fetches & caches the E2E trust token (JWT) shared by all
# three HSBC APIs. Token lives ~60s, so we refresh just before expiry.
# Thread-safe: HSBC calls run in a threadpool (asyncio.to_thread), so a
# double-checked lock ensures only one thread refreshes at a time.
# ---------------------------------------------------------------------------

class _TokenManager:
    def __init__(self, session: requests.Session) -> None:
        self._session = session
        self._token: Optional[str] = None
        self._expires_at: float = 0.0
        self._lock = threading.Lock()

    def _is_valid(self) -> bool:
        # Valid if we have a token and it won't expire within the leeway window.
        return bool(self._token) and (
            self._expires_at - config.HSBC_TOKEN_REFRESH_LEEWAY_SEC > time.time()
        )

    def get_token(self, *, force_refresh: bool = False) -> str:
        if not force_refresh and self._is_valid():
            return self._token  # type: ignore[return-value]

        with self._lock:
            # Double-check inside the lock — another thread may have refreshed.
            if not force_refresh and self._is_valid():
                return self._token  # type: ignore[return-value]
            self._refresh_locked()
            return self._token  # type: ignore[return-value]

    def _refresh_locked(self) -> None:
        if not config.HSBC_TOKEN_USERNAME or not config.HSBC_TOKEN_PASSWORD:
            raise HSBCError(
                "Token credentials missing (HSBC_TOKEN_USERNAME/HSBC_TOKEN_PASSWORD)",
                status_code=500,
            )

        body = {
            "input_token_state": {
                "token_type": "CREDENTIAL",
                "username":   config.HSBC_TOKEN_USERNAME,
                "password":   config.HSBC_TOKEN_PASSWORD,
            },
            "output_token_state": {
                "token_type": "JWT",
            },
        }

        log.info("HSBC token -> refreshing (translator)")
        try:
            resp = self._session.post(
                config.HSBC_TOKEN_URL,
                json=body,
                headers={"Content-Type": "application/json", "Accept": "application/json"},
                timeout=(config.HTTP_CONNECT_TIMEOUT, config.HTTP_READ_TIMEOUT),
            )
        except requests.exceptions.RequestException as e:
            log.error("HSBC token network error: %s", e)
            raise HSBCError(f"Token translator network error: {e}", status_code=502) from e

        if not resp.ok:
            log.error("HSBC token failed status=%d body=%s", resp.status_code, resp.text[:500])
            raise HSBCError(
                f"Token translator returned {resp.status_code}",
                status_code=502, upstream_status=resp.status_code,
            )

        token = self._extract_token(resp)
        if not token:
            raise HSBCError("Token translator response did not contain a JWT", status_code=502)

        self._token = token
        self._expires_at = time.time() + config.HSBC_TOKEN_TTL_SEC
        log.info("HSBC token <- ok (ttl=%ds)", config.HSBC_TOKEN_TTL_SEC)

    @staticmethod
    def _extract_token(resp: requests.Response) -> Optional[str]:
        """Pull the JWT out of the translator response, tolerating shapes:
        - JSON: {"access_token": "..."} / {"token": "..."} / {"issued_token": "..."} /
                {"output_token": "..."} / nested {"output_token_state": {"token": "..."}}
        - Plain text body that is the JWT itself.
        """
        try:
            data = resp.json()
        except ValueError:
            text = resp.text.strip()
            return text or None

        if isinstance(data, dict):
            for key in ("access_token", "token", "issued_token", "output_token", "jwt", "id_token"):
                v = data.get(key)
                if isinstance(v, str) and v:
                    return v
            nested = data.get("output_token_state")
            if isinstance(nested, dict):
                for key in ("token", "access_token", "issued_token", "jwt"):
                    v = nested.get(key)
                    if isinstance(v, str) and v:
                        return v
        elif isinstance(data, str) and data:
            return data
        return None


# ---------------------------------------------------------------------------
# Client
# ---------------------------------------------------------------------------

class HSBCHubClient:
    """One session, three endpoints. Each gets its own trust token from config."""

    def __init__(self) -> None:
        self._validate_config()
        self._session = self._build_session()
        self._tokens = _TokenManager(self._session)

    @staticmethod
    def _validate_config() -> None:
        checks = {
            "HSBC_TRANSACTION_HISTORY_URL": config.HSBC_TRANSACTION_HISTORY_URL,
            "HSBC_APPLY_HOLD_URL":          config.HSBC_APPLY_HOLD_URL,
            "HSBC_DEMAND_DEPOSIT_URL":      config.HSBC_DEMAND_DEPOSIT_URL,
            "HSBC_TOKEN_URL":               config.HSBC_TOKEN_URL,
        }

        for name, url in checks.items():
            log.info("CONFIG %s = %s", name, url or "<EMPTY>")

        errors = []
        for name, url in checks.items():
            if not url:
                errors.append(f"{name} is empty")
                continue
            if "?" in url:
                errors.append(f"{name} contains '?' — query params must not be in the base URL. Got: {url}")
            if any(bad in url for bad in ("accountNumber", "hubCustomerNumber")):
                errors.append(f"{name} has a path variable baked in — base URL must end at the endpoint path. Got: {url}")

        if errors:
            for e in errors:
                log.error("CONFIG ERROR: %s", e)
            raise RuntimeError("HSBCHubClient misconfigured:\n" + "\n".join(errors))

    @staticmethod
    def _build_session() -> requests.Session:
        s = requests.Session()
        s.verify = False  # HSBC internal CA, intentional

        retry = Retry(
            total=config.HTTP_MAX_RETRIES,
            backoff_factor=0.5,
            status_forcelist=(502, 503, 504),
            allowed_methods=frozenset(["GET", "POST"]),
            raise_on_status=False,
        )
        adapter = HTTPAdapter(
            max_retries=retry,
            pool_connections=20,
            pool_maxsize=50,
        )
        s.mount("http://",  adapter)
        s.mount("https://", adapter)

        log.info(
            "HSBCHubClient session ready | verify=False "
            "connect_timeout=%.0fs read_timeout=%.0fs retries=%d",
            config.HTTP_CONNECT_TIMEOUT,
            config.HTTP_READ_TIMEOUT,
            config.HTTP_MAX_RETRIES,
        )
        return s

    def close(self) -> None:
        self._session.close()

    # ------------------------------------------------------------------
    # Endpoint #1 — Transaction History (GET)
    # ------------------------------------------------------------------

    def get_transaction_history(
        self,
        account_number: str,
        from_date:      Optional[str] = None,
        to_date:        Optional[str] = None,
        sensitive_data: str = "",
    ) -> dict[str, Any]:
        params: dict[str, str] = {"accountNumber": account_number}
        if from_date:
            params["fromDate"] = from_date
        if to_date:
            params["toDate"] = to_date

        # sensitive_data is always passed from the request body (JSON) by main.py.
        # Format: '[{"key":"<accountNumber>","value":"<real HSBC account>"}]'
        return self._do(
            method="GET",
            url=config.HSBC_TRANSACTION_HISTORY_URL,
            headers_factory=lambda tok: self._headers_transaction_history(sensitive_data, tok),
            params=params,
            op="transaction_history",
        )

    # ------------------------------------------------------------------
    # Endpoint #2 — Apply Account Hold (POST)
    # ------------------------------------------------------------------

    def apply_account_hold(self, payload: dict[str, Any]) -> dict[str, Any]:
        body = payload if "applyHoldDetails" in payload else {"applyHoldDetails": payload}

        return self._do(
            method="POST",
            url=config.HSBC_APPLY_HOLD_URL,
            headers_factory=lambda tok: self._headers_apply_hold(tok),
            json_body=body,
            op="apply_hold",
        )

    # ------------------------------------------------------------------
    # Endpoint #3 — Demand Deposit Account Enquiry (GET)
    # ------------------------------------------------------------------

    def get_demand_deposit_account(
        self,
        account_number: str,
        sensitive_data: str = "",
    ) -> dict[str, Any]:
        # sensitive_data is always passed from the request body (JSON) by main.py.
        # Format: '[{"key":"<accountNumber>","value":"<real HSBC account>"}]'
        # accountFormat is static per the HSBC contract — always "H".
        params = {
            "accountNumber": account_number,
            "accountFormat": "H",
        }
        return self._do(
            method="GET",
            url=config.HSBC_DEMAND_DEPOSIT_URL,
            headers_factory=lambda tok: self._headers_demand_deposit(sensitive_data, tok),
            params=params,
            op="demand_deposit",
        )

    # ------------------------------------------------------------------
    # Core dispatch
    # ------------------------------------------------------------------

    def _do(
        self,
        *,
        method: str,
        url: str,
        headers_factory,
        op: str,
        params:    Optional[dict[str, Any]] = None,
        json_body: Optional[dict[str, Any]] = None,
        _retry_on_401: bool = True,
    ) -> dict[str, Any]:
        # Fetch (cached) token and build headers fresh for this attempt.
        token = self._tokens.get_token()
        headers = headers_factory(token)

        safe_url = f"{url}?{urlencode(params)}" if params else url
        log.info("HSBC -> %s %s", method, safe_url)

        try:
            resp = self._session.request(
                method=method,
                url=url,
                headers=headers,
                params=params,
                json=json_body,
                timeout=(config.HTTP_CONNECT_TIMEOUT, config.HTTP_READ_TIMEOUT),
            )
        except requests.exceptions.ConnectTimeout as e:
            log.error("HSBC %s connect timeout: %s", op, e)
            raise HSBCError("Timed out connecting to HSBC", status_code=504) from e
        except requests.exceptions.ReadTimeout as e:
            log.error("HSBC %s read timeout: %s", op, e)
            raise HSBCError("Timed out reading from HSBC", status_code=504) from e
        except requests.exceptions.SSLError as e:
            log.error("HSBC %s SSL error: %s", op, e)
            raise HSBCError(f"SSL error: {e}", status_code=502) from e
        except requests.exceptions.RequestException as e:
            log.error("HSBC %s network error: %s", op, e)
            raise HSBCError(f"Network error: {e}", status_code=502) from e

        # Token may have expired mid-flight — refresh once and retry.
        if resp.status_code in (401, 403) and _retry_on_401:
            log.warning("HSBC %s got %d — forcing token refresh & retrying once",
                        op, resp.status_code)
            self._tokens.get_token(force_refresh=True)
            return self._do(
                method=method, url=url, headers_factory=headers_factory, op=op,
                params=params, json_body=json_body, _retry_on_401=False,
            )

        log.info("HSBC <- %s %s status=%d", method, op, resp.status_code)

        try:
            data = resp.json()
        except ValueError:
            data = {"raw": resp.text}

        if not resp.ok:
            log.error("HSBC %s error body: %s", op, resp.text[:2000])
            raise HSBCError(
                f"HSBC {op} returned {resp.status_code}",
                status_code=resp.status_code if 400 <= resp.status_code < 600 else 502,
                upstream_status=resp.status_code,
                upstream_body=data,
            )

        return data
