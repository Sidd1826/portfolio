"""
hsbc_client.py - HSBC Hub API client.

Headers are defined directly in each function (no env vars for headers).
Every request uses the configured proxy and verify=False.
"""
from __future__ import annotations

import uuid
from typing import Any, Optional

import httpx

import config
from logger import get_logger

log = get_logger(__name__)


class HSBCError(Exception):
    def __init__(self, msg: str, status_code: int = 502, upstream_status: Optional[int] = None):
        super().__init__(msg)
        self.status_code     = status_code
        self.upstream_status = upstream_status


class HSBCHubClient:

    def __init__(self, http: httpx.AsyncClient) -> None:
        self._http = http

    # ------------------------------------------------------------------ #
    # 1. Transaction History Enquiry
    #    GET /api/v3/account/demand-deposit-transaction-history
    # ------------------------------------------------------------------ #

    async def get_transaction_history(
        self,
        account_number: str,
        from_date:      Optional[str] = None,   # YYYY-MM-DD
        to_date:        Optional[str] = None,   # YYYY-MM-DD
    ) -> dict[str, Any]:

        headers = {
            "Content-Type":                        "application/json",
            "X-HSBC-CB-Staff-User-Id":             "DMS1",
            "X-HSBC-Chnl-CountryCode":             "IN",
            "X-HSBC-Chnl-Group-Member":            "HSBC",
            "X-HSBC-Client-Id":                    "SBI",
            "X-HSBC-Client-Secret":                "SBI1",
            "X-HSBC-Consumer-Id":                  "INHSBC_HUB_DMS",
            "X-HSBC-E2E-Trust-Token":              "",        # set your token here
            "X-HSBC-GBGF":                         "RBWM",
            "X-HSBC-Global-Channel-Id":            "OHB",
            "X-HSBC-Request-Correlation-Id":       str(uuid.uuid4()),
            "X-HSBC-Session-Correlation-Id":       str(uuid.uuid4()),
            "X-HSBC-Src-UserAgent":               "Mozilla",
            "X-HSBC-Target-System-Environment-Id": "UAT001",
            "X-HSBC-User-Id":                      "user123",
            "X-HSBC-Workstation-Id":               "DMS1",
        }

        params: dict[str, str] = {"accountNumber": account_number}
        if from_date:
            params["fromDate"] = from_date
        if to_date:
            params["toDate"] = to_date

        log.info("GET transaction history | accountNumber=%s", account_number)
        return await self._get(config.TXN_HISTORY_URL, headers=headers, params=params, label="Transaction history")

        headers = {

    # ------------------------------------------------------------------ #
    # Internal helpers — proxy + verify=False on every request
    # ------------------------------------------------------------------ #

    async def _get(
        self,
        url: str,
        headers: dict[str, str],
        params:  dict[str, str],
        label:   str,
    ) -> dict[str, Any]:
        try:
            resp = await self._http.get(url, headers=headers, params=params)
        except httpx.HTTPError as e:
            log.exception("%s network error", label)
            raise HSBCError(f"{label} network error: {type(e).__name__}") from e
        log.info("%s response: HTTP %d", label, resp.status_code)
        return self._parse(resp, label)

    async def _post(
        self,
        url:     str,
        headers: dict[str, str],
        body:    dict[str, Any],
        label:   str,
    ) -> dict[str, Any]:
        try:
            resp = await self._http.post(url, headers=headers, json=body)
        except httpx.HTTPError as e:
            log.exception("%s network error", label)
            raise HSBCError(f"{label} network error: {type(e).__name__}") from e
        log.info("%s response: HTTP %d", label, resp.status_code)
        return self._parse(resp, label)

    @staticmethod
    def _parse(resp: httpx.Response, label: str) -> dict[str, Any]:
        if resp.status_code not in (200, 201):
            log.error("%s failed HTTP %d | body=%s", label, resp.status_code, resp.text[:400])
            raise HSBCError(f"{label} returned HTTP {resp.status_code}", upstream_status=resp.status_code)
        try:
            return resp.json()
        except ValueError:
            return {"raw": resp.text}
