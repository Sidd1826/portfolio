"""
hsbc_client.py — Sync requests client for HSBC Hub APIs.

Three endpoints, each with its own hardcoded header set (per the curls).
Shared concerns: proxy, verify=False, timeouts, retries, error wrapping.

Why sync requests (not httpx async)?
  - HSBC Hub hosts behave more predictably with the corporate proxy + custom CA
    chain via `requests` + urllib3. We run these in a threadpool from FastAPI
    (see main.py) so the event loop stays free.
"""
from __future__ import annotations

import json
from typing import Any, Optional
from urllib.parse import urlencode

import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

import config
from logger import get_logger

log = get_logger(__name__)

# Suppress the InsecureRequestWarning that fires on every verify=False call.
# HSBC internal hosts use an internal CA we don't bundle; verify=False is
# intentional and gated to *.hsbc hosts via config.
requests.packages.urllib3.disable_warnings(  # type: ignore[attr-defined]
    requests.packages.urllib3.exceptions.InsecureRequestWarning  # type: ignore[attr-defined]
)


# ---------------------------------------------------------------------------
# Errors
# ---------------------------------------------------------------------------

class HSBCError(Exception):
    """Raised when an HSBC Hub upstream call fails.

    `status_code`     -> HTTP status we should surface to *our* client
    `upstream_status` -> the actual status returned by HSBC (None on network error)
    """
    def __init__(self, message: str, *, status_code: int = 502, upstream_status: Optional[int] = None):
        super().__init__(message)
        self.status_code = status_code
        self.upstream_status = upstream_status


# ---------------------------------------------------------------------------
# Shared constants (the JWT trust token + correlation IDs are static per
# your instruction — these never change at runtime)
# ---------------------------------------------------------------------------

# NOTE: This JWT is the dev/UAT static trust token from the curls.
# Move to Secret Manager / env var before prod-prod.
_E2E_TRUST_TOKEN = config.HSBC_E2E_TRUST_TOKEN

_STATIC_REQUEST_CORRELATION_ID = config.HSBC_REQUEST_CORRELATION_ID
_STATIC_SESSION_CORRELATION_ID = config.HSBC_SESSION_CORRELATION_ID


# ---------------------------------------------------------------------------
# Client
# ---------------------------------------------------------------------------

class HSBCHubClient:
    """One session, three endpoints. Reuse keepalive connections across calls."""

    def __init__(self) -> None:
        self._session = self._build_session()

    # ------------------------------------------------------------------
    # Session
    # ------------------------------------------------------------------

    @staticmethod
    def _build_session() -> requests.Session:
        s = requests.Session()
        s.verify = False  # HSBC internal CA, intentional

        if config.PROXY_URL:
            s.proxies = {"http": config.PROXY_URL, "https": config.PROXY_URL}

        retry = Retry(
            total=config.HTTP_MAX_RETRIES,
            backoff_factor=0.5,
            status_forcelist=(502, 503, 504),
            allowed_methods=frozenset(["GET", "POST"]),
            raise_on_status=False,
        )
        adapter = HTTPAdapter(
            max_retries=retry,
            pool_connections=20,
            pool_maxsize=50,
        )
        s.mount("http://",  adapter)
        s.mount("https://", adapter)
        return s

    def close(self) -> None:
        self._session.close()

    # ------------------------------------------------------------------
    # Endpoint #1 — Transaction History (GET)
    # ------------------------------------------------------------------

    def get_transaction_history(
        self,
        account_number: str,
        from_date: Optional[str] = None,
        to_date:   Optional[str] = None,
    ) -> dict[str, Any]:
        params: dict[str, str] = {"accountNumber": account_number}
        if from_date:
            params["fromDate"] = from_date
        if to_date:
            params["toDate"] = to_date

        # X-HSBC-Sensitive-Data uses the accountNumber as the value of the
        # ad-hoc key (matches the curl: [{"key":"abc123","value":"INHSBC..."}]).
        sensitive = json.dumps([{"key": account_number, "value": account_number}], separators=(",", ":"))

        return self._do(
            method="GET",
            url=config.HSBC_TRANSACTION_HISTORY_URL,
            headers=self._headers_transaction_history(sensitive),
            params=params,
            op="transaction_history",
        )

    # ------------------------------------------------------------------
    # Endpoint #2 — Apply Account Hold (POST)
    # Body is pass-through: whatever main.py hands us, we forward.
    # ------------------------------------------------------------------

    def apply_account_hold(self, payload: dict[str, Any]) -> dict[str, Any]:
        # Caller may pass either the inner dict or the wrapped {"applyHoldDetails": ...}.
        # Hub expects the wrapped form.
        body = payload if "applyHoldDetails" in payload else {"applyHoldDetails": payload}

        return self._do(
            method="POST",
            url=config.HSBC_APPLY_HOLD_URL,
            headers=self._headers_apply_hold(),
            json_body=body,
            op="apply_hold",
        )

    # ------------------------------------------------------------------
    # Endpoint #3 — Demand Deposit Account Enquiry (GET)
    # ------------------------------------------------------------------

    def get_demand_deposit_account(
        self,
        hub_customer_number: str,
        sensitive_data: Optional[str] = None,
    ) -> dict[str, Any]:
        # If caller didn't supply sensitive-data header, build a default from
        # hubCustomerNumber (matches the curl shape).
        sensitive = sensitive_data or json.dumps(
            [{"key": hub_customer_number, "value": hub_customer_number}],
            separators=(",", ":"),
        )

        return self._do(
            method="GET",
            url=config.HSBC_DEMAND_DEPOSIT_URL,
            headers=self._headers_demand_deposit(sensitive),
            params={"hubCustomerNumber": hub_customer_number},
            op="demand_deposit",
        )

    # ------------------------------------------------------------------
    # Core dispatch — every call goes through here
    # ------------------------------------------------------------------

    def _do(
        self,
        *,
        method: str,
        url: str,
        headers: dict[str, str],
        op: str,
        params:    Optional[dict[str, Any]] = None,
        json_body: Optional[dict[str, Any]] = None,
    ) -> dict[str, Any]:
        # Log the URL with query string but never the trust token / secrets.
        safe_url = f"{url}?{urlencode(params)}" if params else url
        log.info("HSBC -> %s %s", method, safe_url)

        try:
            resp = self._session.request(
                method=method,
                url=url,
                headers=headers,
                params=params,
                json=json_body,
                timeout=(config.HTTP_CONNECT_TIMEOUT, config.HTTP_READ_TIMEOUT),
            )
        except requests.exceptions.ProxyError as e:
            log.error("HSBC %s proxy error: %s", op, e)
            raise HSBCError(f"Proxy error reaching HSBC: {e}", status_code=502) from e
        except requests.exceptions.ConnectTimeout as e:
            log.error("HSBC %s connect timeout: %s", op, e)
            raise HSBCError("Timed out connecting to HSBC", status_code=504) from e
        except requests.exceptions.ReadTimeout as e:
            log.error("HSBC %s read timeout: %s", op, e)
            raise HSBCError("Timed out reading from HSBC", status_code=504) from e
        except requests.exceptions.SSLError as e:
            log.error("HSBC %s SSL error: %s", op, e)
            raise HSBCError(f"SSL error: {e}", status_code=502) from e
        except requests.exceptions.RequestException as e:
            log.error("HSBC %s network error: %s", op, e)
            raise HSBCError(f"Network error: {e}", status_code=502) from e

        log.info("HSBC <- %s %s status=%d", method, op, resp.status_code)

        # Try to parse JSON regardless of status — Hub returns structured error bodies.
        try:
            data = resp.json()
        except ValueError:
            data = {"raw": resp.text}

        if not resp.ok:
            raise HSBCError(
                f"HSBC {op} returned {resp.status_code}",
                status_code=resp.status_code if 400 <= resp.status_code < 600 else 502,
                upstream_status=resp.status_code,
            )

        return data
