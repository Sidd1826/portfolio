# ---------------------------------------------------------------------------
# Lifespan — build shared httpx client with proxy + verify=False
# ---------------------------------------------------------------------------

@asynccontextmanager
async def lifespan(app: FastAPI):
    log.info("Starting HSBC MHA Gateway")

    proxy = config.PROXY_URL or None  # trailing slash already stripped by config.py

    if proxy:
        log.info("PROXY MODE   : ENABLED -> %s", proxy)
    else:
        log.info("PROXY MODE   : DISABLED (set PROXY_URL in .env to enable)")
    log.info("TLS VERIFY   : False")
    log.info("READ TIMEOUT : %.0f seconds", config.HTTP_READ_TIMEOUT)

    # Pass proxy + verify=False directly on AsyncClient — simplest and most reliable.
    # Do NOT use mounts= alongside this; mounts can silently override verify.
    client_kwargs = {
        "verify":  False,
        "timeout": httpx.Timeout(
            connect=config.HTTP_CONNECT_TIMEOUT,
            read=config.HTTP_READ_TIMEOUT,
            write=config.HTTP_READ_TIMEOUT,
            pool=config.HTTP_CONNECT_TIMEOUT,
        ),
        "limits":  httpx.Limits(max_connections=100, max_keepalive_connections=20),
    }
    if proxy:
        client_kwargs["proxy"] = proxy

    http = httpx.AsyncClient(**client_kwargs)

    cipher         = AESCipher(config.UPI_AES_KEY_HEX)
    app.state.upi  = UPIClient(http=http, cipher=cipher)
    app.state.hsbc = HSBCHubClient(http=http)
    app.state.http = http

    log.info("Gateway ready")
    yield
    await http.aclose()
    log.info("Gateway shut down")


# ---------------------------------------------------------------------------
# App
# ---------------------------------------------------------------------------

app = FastAPI(title="HSBC MHA Gateway", version="1.0.0", lifespan=lifespan, docs_url="/docs")


@app.exception_handler(Exception)
async def _unhandled(_req: Request, exc: Exception):
    log.exception("Unhandled: %s", exc)
    return JSONResponse({"error": "internal_error", "message": "An unexpected error occurred"}, status_code=500)

# ---------------------------------------------------------------------------
# Debug — test proxy + connectivity (remove in production)
# ---------------------------------------------------------------------------

@app.get("/debug/proxy", tags=["Debug"])
async def debug_proxy(request: Request):
    """
    Hit this endpoint to check:
    1. What proxy is configured
    2. Whether the proxy can actually reach HSBC
    Call: GET /debug/proxy
    """
    import config as cfg
    http: httpx.AsyncClient = request.app.state.http

    result = {
        "proxy_url":    cfg.PROXY_URL or "NOT SET",
        "verify":       False,
        "read_timeout": cfg.HTTP_READ_TIMEOUT,
        "connectivity": {}
    }

    # Test each HSBC host with a short timeout so it fails fast
    test_urls = {
        "txn_history": cfg.TXN_HISTORY_URL,
        "acct_hold":   cfg.ACCT_HOLD_URL,
        "dda":         cfg.DDA_URL,
    }

    for name, url in test_urls.items():
        try:
            # HEAD request — just check if host is reachable, don't wait for full response
            resp = await http.head(url, timeout=10)
            result["connectivity"][name] = {"status": "reachable", "http_code": resp.status_code}
        except httpx.ProxyError as e:
            result["connectivity"][name] = {"status": "proxy_error", "detail": str(e)}
        except httpx.ConnectError as e:
            result["connectivity"][name] = {"status": "connect_error", "detail": str(e)}
        except httpx.ConnectTimeout:
            result["connectivity"][name] = {"status": "connect_timeout", "detail": "Could not reach host within 10s"}
        except httpx.ReadTimeout:
            # Got through proxy but server didn't respond — proxy is working
            result["connectivity"][name] = {"status": "proxy_ok_server_slow", "detail": "Proxy connected but server timed out — proxy is working"}
        except Exception as e:
            result["connectivity"][name] = {"status": "error", "detail": str(e)}

    return result


# ---------------------------------------------------------------------------
# UPI Secure
# ---------------------------------------------------------------------------

@app.post("/api/v1/upi/secure", tags=["UPI"])
async def upi_secure(
    request: Request,
    payload: dict[str, Any] = Body(..., examples=[{"upiId": "user@bank", "amount": 100.5}]),
):
    """Encrypt payload -> call HSBC UPI -> decrypt response."""
    req_id = str(uuid.uuid4())
    log.info("[%s] POST /upi/secure", req_id)
    upi: UPIClient = request.app.state.upi
    try:
        result = await upi.send_secure(payload)
        return {"requestId": req_id, "data": result}
    except CryptoError as e:
        log.error("[%s] crypto: %s", req_id, e)
        raise HTTPException(502, detail={"requestId": req_id, "error": "crypto_error", "message": str(e)})
    except UPIError as e:
        log.error("[%s] upi: %s", req_id, e)
        raise HTTPException(e.status_code, detail={"requestId": req_id, "error": "upstream_error",
                                                    "message": str(e), "upstreamStatus": e.upstream_status})

# Apply Account Hold
# ---------------------------------------------------------------------------

class ApplyHoldDetails(BaseModel):
    accountNumber:        str           = Field(..., description="e.g. INHSBC011019072006")
    accountCurrencyCode:  str           = Field(..., description="e.g. INR")
    holdTillExpiredDate:  str           = Field(..., description="YYYY-MM-DD")
    holdType:             str           = Field(..., description="e.g. B")
    holdCurrency:         str           = Field(..., description="e.g. INR")
    holdAmount:           str           = Field(..., description="e.g. 28.28")
    customerNumber:       Optional[str] = None
    globalUserId:         Optional[str] = None
    hsbcnetCustomerId:    Optional[str] = None
    holdNarrative1:       Optional[str] = None
    holdNarrative2:       Optional[str] = None
    holdNarrative3:       Optional[str] = None
    holdNarrative4:       Optional[str] = None
    sequenceNumber:       Optional[str] = None
    holdValueDate:        Optional[str] = None
    holdDepositDate:      Optional[str] = None
    clearingArea:         Optional[str] = None
    clearingNumber:       Optional[str] = None
    balanceCheckRequired: Optional[str] = None
    onlineOfflineFlag:    Optional[str] = None
    debitCardNumber:      Optional[str] = None

class ApplyHoldRequest(BaseModel):
    applyHoldDetails: ApplyHoldDetails


@app.post("/api/v1/accounts/apply-hold", tags=["Accounts"])
async def apply_account_hold(request: Request, body: ApplyHoldRequest = Body(...)):
    """Apply a hold on an HSBC account."""
    req_id = str(uuid.uuid4())
    log.info("[%s] POST /accounts/apply-hold | accountNumber=%s amount=%s",
             req_id, body.applyHoldDetails.accountNumber, body.applyHoldDetails.holdAmount)
    hub: HSBCHubClient = request.app.state.hsbc
    try:
        result = await hub.apply_account_hold(body.applyHoldDetails.model_dump(exclude_none=True))
        return {"requestId": req_id, "data": result}
    except HSBCError as e:
        log.error("[%s] %s", req_id, e)
        raise HTTPException(e.status_code, detail={"requestId": req_id, "error": "upstream_error",
                                                    "message": str(e), "upstreamStatus": e.upstream_status})
