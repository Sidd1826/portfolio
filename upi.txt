"""
upi.py - AES-128-ECB cipher and HSBC UPI client.
"""
from __future__ import annotations

import asyncio
import base64
import binascii
import json
import time
from dataclasses import dataclass
from typing import Any, Optional

import httpx
from cryptography.hazmat.primitives import padding
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes

import config
from logger import get_logger

log = get_logger(__name__)


# ---------------------------------------------------------------------------
# Exceptions
# ---------------------------------------------------------------------------

class CryptoError(Exception):
    pass

class UPIError(Exception):
    def __init__(self, msg: str, status_code: int = 502, upstream_status: Optional[int] = None):
        super().__init__(msg)
        self.status_code     = status_code
        self.upstream_status = upstream_status


# ---------------------------------------------------------------------------
# AES-128-ECB  (Java: AES/ECB/PKCS5Padding compatible)
# ---------------------------------------------------------------------------

class AESCipher:
    KEY_BYTES   = 16
    BLOCK_BITS  = 128
    BLOCK_BYTES = 16

    def __init__(self, key_hex: str) -> None:
        if not key_hex:
            raise CryptoError("UPI_AES_KEY_HEX is not set")
        try:
            key = bytes.fromhex(key_hex.strip())
        except ValueError as e:
            raise CryptoError(f"UPI_AES_KEY_HEX is not valid hex: {e}") from e
        if len(key) != self.KEY_BYTES:
            raise CryptoError(f"AES key must be {self.KEY_BYTES} bytes, got {len(key)}")
        self._key = key

    def encrypt(self, payload: dict[str, Any]) -> str:
        pt      = json.dumps(payload, separators=(",", ":"), ensure_ascii=False).encode()
        padder  = padding.PKCS7(self.BLOCK_BITS).padder()
        padded  = padder.update(pt) + padder.finalize()
        enc     = Cipher(algorithms.AES(self._key), modes.ECB()).encryptor()
        return (enc.update(padded) + enc.finalize()).hex().upper()

    def decrypt(self, hex_ct: str) -> dict[str, Any]:
        cleaned = "".join(hex_ct.split())
        try:
            ct = bytes.fromhex(cleaned)
        except (ValueError, binascii.Error) as e:
            raise CryptoError(f"Not valid hex: {e}") from e
        if not ct or len(ct) % self.BLOCK_BYTES != 0:
            raise CryptoError("Ciphertext length invalid")
        try:
            dec      = Cipher(algorithms.AES(self._key), modes.ECB()).decryptor()
            padded   = dec.update(ct) + dec.finalize()
            unpadder = padding.PKCS7(self.BLOCK_BITS).unpadder()
            plain    = unpadder.update(padded) + unpadder.finalize()
        except ValueError as e:
            raise CryptoError("Decryption failed (bad padding or wrong key)") from e
        try:
            return json.loads(plain.decode("utf-8"))
        except (UnicodeDecodeError, json.JSONDecodeError) as e:
            raise CryptoError(f"Decrypted bytes not valid JSON: {e}") from e


# ---------------------------------------------------------------------------
# UPI Client
# ---------------------------------------------------------------------------

@dataclass
class _Token:
    value: str
    expires_at: float

def _jwt_exp(jwt: str) -> Optional[float]:
    try:
        parts = jwt.split(".")
        if len(parts) != 3:
            return None
        b64 = parts[1] + "=" * (-len(parts[1]) % 4)
        return float(json.loads(base64.urlsafe_b64decode(b64)).get("exp", 0)) or None
    except Exception:
        return None


class UPIClient:
    def __init__(self, http: httpx.AsyncClient, cipher: AESCipher) -> None:
        self._http       = http
        self._cipher     = cipher
        self._login_url  = f"{config.UPI_BASE_URL.rstrip('/')}{config.UPI_LOGIN_PATH}"
        self._secure_url = f"{config.UPI_BASE_URL.rstrip('/')}{config.UPI_SECURE_PATH}"
        self._token: Optional[_Token] = None
        self._lock       = asyncio.Lock()

    async def send_secure(self, payload: dict[str, Any]) -> dict[str, Any]:
        encrypted     = self._cipher.encrypt(payload)
        response_body = await self._post_secure(encrypted, retry_on_401=True)
        for key in ("responseMsg", "response_msg", "data", "result"):
            v = response_body.get(key)
            if isinstance(v, str) and v:
                return self._cipher.decrypt(v)
        raise UPIError(f"No encrypted field in response. Keys: {list(response_body.keys())}")

    async def _get_token(self) -> str:
        tok = self._token
        if tok and tok.expires_at - config.TOKEN_REFRESH_LEEWAY > time.time():
            return tok.value
        async with self._lock:
            tok = self._token
            if tok and tok.expires_at - config.TOKEN_REFRESH_LEEWAY > time.time():
                return tok.value
            log.info("Fetching UPI auth token")
            value = await self._login()
            self._token = _Token(value=value, expires_at=_jwt_exp(value) or time.time() + config.TOKEN_DEFAULT_TTL)
            return value

    async def _login(self) -> str:
        try:
            resp = await self._http.post(
                self._login_url,
                json={"username": config.UPI_USERNAME, "password": config.UPI_PASSWORD},
                headers={"Content-Type": "application/json"},
            )
        except httpx.HTTPError as e:
            raise UPIError(f"Login network error: {type(e).__name__}") from e
        if resp.status_code != 200:
            raise UPIError(f"Login failed HTTP {resp.status_code}", upstream_status=resp.status_code)
        data = resp.json()
        for src in [data, data.get("data") or {}]:
            if isinstance(src, dict):
                for k in ("token", "access_token", "accessToken", "authToken"):
                    v = src.get(k)
                    if isinstance(v, str) and v:
                        return v
        raise UPIError("No token in login response")

    async def _post_secure(self, hex_payload: str, retry_on_401: bool) -> dict[str, Any]:
        token = await self._get_token()
        try:
            resp = await self._http.post(
                self._secure_url,
                json={"requestMsg": hex_payload},
                headers={"Authorization": f"Bearer {token}", "Content-Type": "application/json"},
            )
        except httpx.HTTPError as e:
            raise UPIError(f"Secure endpoint network error: {type(e).__name__}") from e
        if resp.status_code == 401 and retry_on_401:
            log.warning("401 on secure call — refreshing token")
            async with self._lock:
                self._token = None
            return await self._post_secure(hex_payload, retry_on_401=False)
        if resp.status_code != 200:
            raise UPIError(f"Secure endpoint HTTP {resp.status_code}", upstream_status=resp.status_code)
        return resp.json()
