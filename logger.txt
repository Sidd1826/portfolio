"""
logger.py - Centralised logging.

Writes to console + rotating log file.
Import `get_logger(__name__)` in every module.
"""
import logging
import os
from logging.handlers import RotatingFileHandler

_LOG_LEVEL   = os.getenv("LOG_LEVEL", "INFO").upper()
_LOG_DIR     = os.getenv("LOG_DIR",   "logs")
_LOG_FILE    = os.path.join(_LOG_DIR, "hsbc_gateway.log")
_FMT         = "%(asctime)s [%(levelname)-8s] %(name)s - %(message)s"
_DATE_FMT    = "%Y-%m-%d %H:%M:%S"
_initialized = False


def _setup() -> None:
    global _initialized
    if _initialized:
        return
    _initialized = True

    os.makedirs(_LOG_DIR, exist_ok=True)

    root = logging.getLogger()
    root.setLevel(_LOG_LEVEL)

    fmt = logging.Formatter(_FMT, datefmt=_DATE_FMT)

    # Console handler
    ch = logging.StreamHandler()
    ch.setFormatter(fmt)
    root.addHandler(ch)

    # Rotating file handler (10 MB x 5 backups)
    fh = RotatingFileHandler(_LOG_FILE, maxBytes=10 * 1024 * 1024, backupCount=5, encoding="utf-8")
    fh.setFormatter(fmt)
    root.addHandler(fh)

    # Quiet noisy libs
    logging.getLogger("uvicorn.access").setLevel(logging.WARNING)
    logging.getLogger("httpx").setLevel(logging.WARNING)
    logging.getLogger("httpcore").setLevel(logging.WARNING)


def get_logger(name: str) -> logging.Logger:
    _setup()
    return logging.getLogger(name)
