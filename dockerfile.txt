FROM nexus3.systems.uk.hsbc:18096/com/hsbc/group/itid/es/dc/ubuntu/gcr-ubuntu-2404:latest

ENV APP_HOME=/app
USER root
WORKDIR $APP_HOME

COPY pip.conf /etc/pip.conf
COPY secret_nexus3 /etc/apt/auth.conf
COPY sources.list /etc/apt/sources.list

RUN apt-get update && apt-get -y upgrade \
    && apt-get dist-upgrade -y \
    && apt-get install -y --no-install-recommends \
        python3.12 \
        python3.12-venv \
        curl \
    && apt-get clean \
    && rm -rf /var/lib/apt/lists/*

COPY requirements.txt .
COPY . .

RUN python3.12 -m venv venv \
    && venv/bin/python -m ensurepip --upgrade \
    && venv/bin/python -m pip install --no-cache-dir --upgrade \
        "pip==26.1.2" "setuptools==82.0.0" "wheel" \
    && venv/bin/python -m pip install --no-cache-dir -r requirements.txt

EXPOSE 8000
CMD ["venv/bin/python", "main.py"]
