def get_bq_client() -> bigquery.Client:
    credentials = service_account.Credentials.from_service_account_file(SERVICE_ACCOUNT_FILE)
    return bigquery.Client(credentials=credentials, project=credentials.project_id)


def run_search(filters: dict, limit: int) -> list[dict]:
    """Build a parameterized query from the supplied filters and execute it."""
    try:
        client = get_bq_client()

        # Build WHERE clause with parameterized queries (safe from SQL injection)
        where_clauses = []
        query_params = []

        for field, value in filters.items():
            if field not in ALLOWED_FIELDS:
                # Skip unknown fields defensively
                continue

            # Pick the BigQuery scalar type based on Python type
            if isinstance(value, (int, float)):
                param_type = "FLOAT64" if isinstance(value, float) else "INT64"
            else:
                param_type = "STRING"

            where_clauses.append(f"`{field}` = @{field}")
            query_params.append(bigquery.ScalarQueryParameter(field, param_type, value))

        table_ref = f"`{PROJECT_ID}.{DATASET_ID}.{TABLE_ID}`"
        query = f"SELECT * FROM {table_ref}"
        if where_clauses:
            query += " WHERE " + " AND ".join(where_clauses)
        query += " LIMIT @row_limit"
        query_params.append(bigquery.ScalarQueryParameter("row_limit", "INT64", limit))

        # Job config: parameters + optional KMS encryption
        job_config_kwargs = {"query_parameters": query_params}
        if KMS_KEY:
            job_config_kwargs["destination_encryption_configuration"] = (
                bigquery.EncryptionConfiguration(kms_key_name=KMS_KEY)
            )
        job_config = bigquery.QueryJobConfig(**job_config_kwargs)

        logger.info("Running query: %s | params: %s", query, filters)

        job = client.query(query, job_config=job_config)
        results = [dict(row) for row in job.result()]

        logger.info("Rows fetched: %d", len(results))
        return results

    except Exception as e:
        logger.exception("BigQuery search failed")
        raise HTTPException(
            status_code=500,
            detail=f"Error occurred while fetching data from BigQuery: {str(e)}",
        )


@app.get("/health")
def health():
    return {"status": "ok"}


@app.post("/search")
def search(req: SearchRequest):
    # Extract only the fields the user actually provided (non-None), excluding `limit`
    filters = {
        k: v for k, v in req.model_dump(exclude_none=True).items()
        if k != "limit"
    }

    if not filters:
        raise HTTPException(
            status_code=400,
            detail="Provide at least one field to search on.",
        )

    rows = run_search(filters, req.limit or 100)
    return {
        "filters_used": filters,
        "row_count": len(rows),
        "rows": rows,
    }
