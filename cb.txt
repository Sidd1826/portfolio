import os
import logging
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from google.cloud import bigquery
from google.oauth2 import service_account

# ---------- Config ----------
# Path to service account file - can be absolute or relative to where you run the app
SERVICE_ACCOUNT_FILE = os.getenv("SERVICE_ACCOUNT_FILE", "serviceaccount.json")

# Optional: KMS key for customer-managed encryption (CMEK)
# Format: projects/PROJECT/locations/LOCATION/keyRings/RING/cryptoKeys/KEY
KMS_KEY = os.getenv("KMS_KEY")  # leave unset if not using CMEK

# ---------- Logging ----------
logging.basicConfig(level=logging.DEBUG)
logger = logging.getLogger(__name__)

# ---------- FastAPI ----------
app = FastAPI(title="BigQuery API")


class QueryRequest(BaseModel):
    query: str


def gcp_kms(creds_path: str, query: str) -> list[dict]:
    try:
        credentials = service_account.Credentials.from_service_account_file(creds_path)
        client = bigquery.Client(
            credentials=credentials,
            project=credentials.project_id,
        )

        job_config = None
        if KMS_KEY:
            enc = bigquery.EncryptionConfiguration(kms_key_name=KMS_KEY)
            job_config = bigquery.QueryJobConfig(
                destination_encryption_configuration=enc
            )

        job = client.query(query, job_config=job_config)
        results = [dict(row) for row in job.result()]

        logger.debug("Query executed successfully. Rows fetched: %d", len(results))
        return results

    except Exception as e:
        logger.exception("BigQuery query failed")
        raise HTTPException(
            status_code=500,
            detail=f"Error occurred while fetching data from BigQuery: {str(e)}",
        )


@app.get("/health")
def health():
    return {"status": "ok", "creds_file": SERVICE_ACCOUNT_FILE}


@app.post("/query")
def run_query(req: QueryRequest):
    results = gcp_kms(SERVICE_ACCOUNT_FILE, req.query)
    return {"row_count": len(results), "rows": results}
