import os
import logging
from typing import Optional
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field
from google.cloud import bigquery
from google.oauth2 import service_account

# ---------- Config ----------
SERVICE_ACCOUNT_FILE = os.getenv("SERVICE_ACCOUNT_FILE", "serviceaccount.json")
KMS_KEY = os.getenv("KMS_KEY")  # Optional, leave unset if not using CMEK

PROJECT_ID = os.getenv("PROJECT_ID", "your-gcp-project-id")
DATASET_ID = os.getenv("DATASET_ID", "your_dataset")
TABLE_ID = os.getenv("TABLE_ID", "your_table")

# ---------- Logging ----------
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# ---------- Field type mapping ----------
FIELD_TYPES = {
    "XQCBA1": "STRING",
    "OGBEAC": "STRING",
    "OGBEA1": "STRING",
    "OGBEA2": "STRING",
    "OGBEA3": "STRING",
    "OGBEA4": "STRING",
    "OGT521": "STRING",
    "OGTRNO": "STRING",
    "OGTMZ1": "STRING",
    "OGCPDT": "INT64",
    "OGCPTM": "INT64",
    "OGAVDT": "INT64",
    "OGAVTM": "INT64",
    "OGPYAM": "FLOAT64",
    "OGPALE": "FLOAT64",
    "OGPYCY": "STRING",
    "OGNAR1": "STRING",
    "OGNAR2": "STRING",
    "OGNAR3": "STRING",
    "OGNAR4": "STRING",
}

ALLOWED_FIELDS = set(FIELD_TYPES.keys())

# ---------- FastAPI ----------
app = FastAPI(title="BigQuery Search API")


# ---------- Request Model ----------
class SearchRequest(BaseModel):
    XQCBA1: Optional[str] = Field(None, description="Correspondent Bank/Branch Name and Address")
    OGBEAC: Optional[str] = Field(None, description="Beneficiary A/C No")
    OGBEA1: Optional[str] = Field(None, description="Beneficiary Name and Address Line 1")
    OGBEA2: Optional[str] = Field(None, description="Beneficiary Name and Address Line 2")
    OGBEA3: Optional[str] = Field(None, description="Beneficiary Name and Address Line 3")
    OGBEA4: Optional[str] = Field(None, description="Beneficiary Name and Address Line 4")
    OGT521: Optional[str] = Field(None, description="Tag52 Ordering Institution Line 1")
    OGTRNO: Optional[str] = Field(None, description="Transaction Reference No")
    OGTMZ1: Optional[str] = Field(None, description="Time Stamp")
    OGCPDT: Optional[int] = Field(None, description="Capture Date (e.g. 20260926)")
    OGCPTM: Optional[int] = Field(None, description="Capture Time (e.g. 133101)")
    OGAVDT: Optional[int] = Field(None, description="Approval/Verification Date (e.g. 20260926)")
    OGAVTM: Optional[int] = Field(None, description="Approval/Verification Time (e.g. 133101)")
    OGPYAM: Optional[float] = Field(None, description="Payment Amount")
    OGPALE: Optional[float] = Field(None, description="Lcy Payment Amount")
    OGPYCY: Optional[str] = Field(None, description="Payment Currency")
    OGNAR1: Optional[str] = Field(None, description="Narrative 1")
    OGNAR2: Optional[str] = Field(None, description="Narrative 2")
    OGNAR3: Optional[str] = Field(None, description="Narrative 3")
    OGNAR4: Optional[str] = Field(None, description="Narrative 4")
    limit: Optional[int] = Field(100, ge=1, le=10000, description="Max rows to return")


# ---------- BQ Client ----------
def get_bq_client() -> bigquery.Client:
    credentials = service_account.Credentials.from_service_account_file(SERVICE_ACCOUNT_FILE)
    return bigquery.Client(credentials=credentials, project=PROJECT_ID)


# ---------- Core Query Function ----------
def run_search(filters: dict, limit: int) -> list[dict]:
    try:
        client = get_bq_client()

        where_clauses = []
        query_params = []

        for field, value in filters.items():
            if field not in ALLOWED_FIELDS:
                logger.warning("Skipping unknown field: %s", field)
                continue

            param_type = FIELD_TYPES[field]
            where_clauses.append(f"`{field}` = @{field}")
            query_params.append(
                bigquery.ScalarQueryParameter(field, param_type, value)
            )

        query_params.append(
            bigquery.ScalarQueryParameter("row_limit", "INT64", limit)
        )

        table_ref = f"`{PROJECT_ID}.{DATASET_ID}.{TABLE_ID}`"
        query = f"SELECT * FROM {table_ref}"
        if where_clauses:
            query += " WHERE " + " AND ".join(where_clauses)
        query += " LIMIT @row_limit"

        # Job config
        job_config_kwargs = {"query_parameters": query_params}
        if KMS_KEY:
            job_config_kwargs["destination_encryption_configuration"] = (
                bigquery.EncryptionConfiguration(kms_key_name=KMS_KEY)
            )
        job_config = bigquery.QueryJobConfig(**job_config_kwargs)

        logger.info("Executing query: %s | filters: %s", query, filters)

        job = client.query(query, job_config=job_config)
        results = [dict(row) for row in job.result()]

        logger.info("Rows fetched: %d", len(results))
        return results

    except HTTPException:
        raise
    except Exception as e:
        logger.exception("BigQuery query failed")
        raise HTTPException(
            status_code=500,
            detail=f"Error occurred while fetching data from BigQuery: {str(e)}",
        )


# ---------- Routes ----------
@app.get("/health")
def health():
    return {
        "status": "ok",
        "project": PROJECT_ID,
        "dataset": DATASET_ID,
        "table": TABLE_ID,
    }


@app.post("/search")
def search(req: SearchRequest):
    filters = {
        k: v for k, v in req.model_dump(exclude_none=True).items()
        if k != "limit"
    }

    if not filters:
        raise HTTPException(
            status_code=400,
            detail="Provide at least one field to filter on.",
        )

    rows = run_search(filters, req.limit or 100)
    return {
        "filters_used": filters,
        "row_count": len(rows),
        "rows": rows,
    }
