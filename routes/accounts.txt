"""
routes/accounts.py - Account Hold + Demand Deposit Enquiry.
Caller passes only business data — no HSBC headers needed.
"""
from __future__ import annotations

import uuid
from typing import Optional

from fastapi import APIRouter, Body, HTTPException, Query, Request
from pydantic import BaseModel, Field

from hsbc_client import HSBCError, HSBCHubClient
from logger import get_logger

logger = get_logger(__name__)
router = APIRouter(prefix="/accounts", tags=["Accounts"])


# ---------------------------------------------------------------------------
# Schema
# ---------------------------------------------------------------------------

class ApplyHoldDetails(BaseModel):
    customerNumber:       Optional[str] = None
    globalUserId:         Optional[str] = None 
    accountNumber:        str  = Field(..., description="e.g. INHSBC011019072006")
    hsbcnetCustomerId:    Optional[str] = None
    accountCurrencyCode:  str  = Field(..., description="e.g. INR")
    holdTillExpiredDate:  str  = Field(..., description="YYYY-MM-DD")
    holdType:             str  = Field(..., description="Hold type code e.g. B")
    holdCurrency:         str  = Field(..., description="e.g. INR")
    holdAmount:           str  = Field(..., description="e.g. 28.28")
    holdNarrative1:       Optional[str] = None
    holdNarrative2:       Optional[str] = None
    holdNarrative3:       Optional[str] = None
    holdNarrative4:       Optional[str] = None
    sequenceNumber:       Optional[str] = None
    holdValueDate:        Optional[str] = Field(default=None, description="YYYY-MM-DD")
    holdDepositDate:      Optional[str] = Field(default=None, description="YYYY-MM-DD")
    clearingArea:         Optional[str] = None
    clearingNumber:       Optional[str] = None
    balanceCheckRequired: Optional[str] = None
    onlineOfflineFlag:    Optional[str] = None
    debitCardNumber:      Optional[str] = None


class ApplyHoldRequest(BaseModel):
    applyHoldDetails: ApplyHoldDetails


# ---------------------------------------------------------------------------
# Route 1 — Apply Account Hold
# ---------------------------------------------------------------------------

@router.post("/apply-hold", summary="Apply a hold on an account")
async def apply_account_hold(
    request: Request,
    body: ApplyHoldRequest = Body(...),
):
    """
    Apply a hold on an HSBC account.
    Just pass the hold details — all HSBC headers are managed internally.
    """
    req_id = str(uuid.uuid4())
    logger.info(
        "[%s] POST /accounts/apply-hold | accountNumber=%s amount=%s",
        req_id, body.applyHoldDetails.accountNumber, body.applyHoldDetails.holdAmount,
    )

    hub: HSBCHubClient = request.app.state.hsbc
    try:
        result = await hub.apply_account_hold(
            apply_hold_details=body.applyHoldDetails.model_dump(exclude_none=True)
        )
        logger.info("[%s] Apply account hold success", req_id)
        return {"requestId": req_id, "data": result}
    except HSBCError as e:
        logger.error("[%s] %s | upstream=%s", req_id, e, e.upstream_status)
        raise HTTPException(status_code=e.status_code, detail={
            "requestId": req_id, "error": "upstream_error",
            "message": str(e), "upstreamStatus": e.upstream_status,
        })


# ---------------------------------------------------------------------------
# Route 2 — Demand Deposit Account Enquiry
# ---------------------------------------------------------------------------

@router.get("/demand-deposit", summary="Demand deposit account enquiry")
async def get_demand_deposit_account(
    request: Request,
    hub_customer_number: str           = Query(..., alias="hubCustomerNumber", description="e.g. 1111"),
    sensitive_data:      Optional[str] = Query(default=None, alias="sensitiveData",
                                               description='e.g. [{"key":"1111","value":"INHSBC003297637"}]'),
):
    """
    Fetch demand deposit account details.
    Just pass `hubCustomerNumber` — all HSBC headers are managed internally.
    """
    req_id = str(uuid.uuid4())
    logger.info("[%s] GET /accounts/demand-deposit | hubCustomerNumber=%s", req_id, hub_customer_number)

    hub: HSBCHubClient = request.app.state.hsbc
    try:
        result = await hub.get_demand_deposit_account(
            hub_customer_number=hub_customer_number,
            sensitive_data=sensitive_data,
        )
        logger.info("[%s] Demand deposit account enquiry success", req_id)
        return {"requestId": req_id, "data": result}
    except HSBCError as e:
        logger.error("[%s] %s | upstream=%s", req_id, e, e.upstream_status)
        raise HTTPException(status_code=e.status_code, detail={
            "requestId": req_id, "error": "upstream_error",
            "message": str(e), "upstreamStatus": e.upstream_status,
        })
