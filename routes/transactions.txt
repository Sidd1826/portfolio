"""
routes/transactions.py - Transaction History Enquiry.
Caller passes only business params — no HSBC headers needed.
"""
from __future__ import annotations

import uuid
from typing import Optional

from fastapi import APIRouter, HTTPException, Query, Request

from hsbc_client import HSBCError, HSBCHubClient
from logger import get_logger

logger = get_logger(__name__)
router = APIRouter(prefix="/transactions", tags=["Transactions"])


@router.get("/history", summary="Demand deposit transaction history enquiry")
async def get_transaction_history(
    request: Request,
    account_number: str           = Query(..., alias="accountNumber", description="e.g. abc123"),
    from_date:      Optional[str] = Query(default=None, alias="fromDate",  description="YYYY-MM-DD"),
    to_date:        Optional[str] = Query(default=None, alias="toDate",    description="YYYY-MM-DD"),
):
    """
    Fetch transaction history for an account.
    Just pass `accountNumber` — all HSBC headers are managed internally.
    """
    req_id = str(uuid.uuid4())
    logger.info("[%s] GET /transactions/history | accountNumber=%s", req_id, account_number)

    extra: dict[str, str] = {}
    if from_date:
        extra["fromDate"] = from_date
    if to_date:
        extra["toDate"] = to_date

    hub: HSBCHubClient = request.app.state.hsbc
    try:
        result = await hub.get_transaction_history(account_number=account_number, extra_params=extra)
        logger.info("[%s] Transaction history success", req_id)
        return {"requestId": req_id, "data": result}
    except HSBCError as e:
        logger.error("[%s] %s | upstream=%s", req_id, e, e.upstream_status)
        raise HTTPException(status_code=e.status_code, detail={
            "requestId": req_id, "error": "upstream_error",
            "message": str(e), "upstreamStatus": e.upstream_status,
        })
